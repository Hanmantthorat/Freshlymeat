        let cart = [];

        function renderProducts(items) {
            const container = document.getElementById("products-container");
            container.innerHTML = "";

            if (items.length === 0) {
                container.innerHTML = `<div class="col-span-full text-center py-12 text-gray-400 font-semibold">No items match your search.</div>`;
                return;
            }

            items.forEach(product => {
                const card = document.createElement("div");
                card.className = "bg-white rounded-2xl border border-gray-100 shadow-sm hover:shadow-md transition p-4 flex flex-col justify-between";
                card.innerHTML = `
                    <div>
                        <div class="relative h-44 rounded-xl overflow-hidden bg-gray-100 mb-3">
                            <img src="${product.img}" alt="${product.name}" class="w-full h-full object-cover" loading="lazy" onerror="this.onerror=null;this.src='https://images.unsplash.com/photo-1587593810167-a84920ea0781?w=500&q=80';">

                        </div>
                        <div class="flex items-center justify-between text-xs text-amber-500 font-bold mb-1">
                            <span><i class="fa-solid fa-star"></i> ${product.rating}</span>
                        </div>
                        <h3 class="font-bold text-gray-900 text-sm mb-2">${product.name}</h3>
                    </div>
                    <div>
                        <div class="mb-3">
                            <label class="block text-[11px] font-semibold text-gray-500 mb-1">Select Weight:</label>
                            <select id="weight-${product.id}" onchange="updatePriceDisplay(${product.id}, this.value)" class="w-full bg-gray-50 border border-gray-200 text-xs rounded-lg p-2 font-semibold">
                                <option value="0.5">500 grams (₹${Math.round(product.basePrice * 0.5)})</option>
                                <option value="1.0" selected>1.0 kg (₹${product.basePrice})</option>
                                <option value="2.0">2.0 kg (₹${Math.round(product.basePrice * 2.0)})</option>
                            </select>
                        </div>
                        <div class="flex items-center justify-between pt-2 border-t">
                            <span class="text-base font-black text-gray-900" id="price-display-${product.id}">₹${product.basePrice}</span>
                            <button onclick="addToCart(${product.id})" class="bg-brand-600 hover:bg-brand-700 text-white font-bold text-xs px-4 py-2 rounded-xl transition">
                                <i class="fa-solid fa-plus mr-1"></i> Add
                            </button>
                        </div>
                    </div>

                container.appendChild(card);
            });
        }

        function updatePriceDisplay(productId, multiplier) {
            const product = products.find(p => p.id === productId);
            const calculatedPrice = Math.round(product.basePrice * parseFloat(multiplier));
            document.getElementById(`price-display-${productId}`).innerText = `₹${calculatedPrice}`;
        }

        function filterCategory(category, element) {
            activeCategory = category;
            document.querySelectorAll('.category-btn').forEach(btn => {
                btn.className = "category-btn hover:text-brand-600 transition pb-1 whitespace-nowrap text-gray-600";
            });
            if (element) {
                element.className = "category-btn text-brand-600 border-b-2 border-brand-600 pb-1 font-bold whitespace-nowrap";
            }

            const filtered = category === "all" ? products : products.filter(p => p.category === category);
            renderProducts(filtered);
        }

        function handleSearch() {
            const query = (document.getElementById("search-input").value || document.getElementById("mobile-search-input").value).toLowerCase();
            const filtered = products.filter(p => p.name.toLowerCase().includes(query));
            renderProducts(filtered);
        }

        function addToCart(productId) {
            const product = products.find(p => p.id === productId);
            const weightVal = parseFloat(document.getElementById(`weight-${productId}`).value);
            const itemPrice = Math.round(product.basePrice * weightVal);

            const existingIndex = cart.findIndex(item => item.id === productId && item.weight === weightVal);
            if (existingIndex > -1) {
                cart[existingIndex].qty += 1;
            } else {
                cart.push({ id: productId, name: product.name, weight: weightVal, price: itemPrice, qty: 1 });
            }

            updateCartUI();
            toggleCartDrawer(true);
        }

        function updateCartUI() {
            const cartItemsContainer = document.getElementById("cart-items-container");
            const cartBadge = document.getElementById("cart-badge");
            const cartSubtotal = document.getElementById("cart-subtotal");
            const cartTotal = document.getElementById("cart-total");
            const cartTotalHeader = document.getElementById("cart-total-header");
            const cartItemCount = document.getElementById("cart-item-count");

            cartItemsContainer.innerHTML = "";
            let total = 0;
            let totalQty = 0;

            cart.forEach((item, index) => {
                const itemTotal = item.price * item.qty;
                total += itemTotal;
                totalQty += item.qty;

                const cartRow = document.createElement("div");
                cartRow.className = "flex items-center justify-between bg-gray-50 p-3 rounded-xl border border-gray-100";
                cartRow.innerHTML = `
                    <div>
                        <h4 class="font-bold text-xs text-gray-800">${item.name}</h4>
                        <span class="text-[10px] text-gray-500">${item.weight >= 1 ? item.weight + ' kg' : (item.weight * 1000) + ' g'} | ₹${item.price}</span>
                    </div>
                    <div class="flex items-center gap-2">
                        <div class="flex items-center border rounded-lg bg-white">
                            <button onclick="changeQty(${index}, -1)" class="px-2 py-0.5 text-xs text-gray-600 font-bold">-</button>
                            <span class="text-xs px-2 font-bold">${item.qty}</span>
                            <button onclick="changeQty(${index}, 1)" class="px-2 py-0.5 text-xs text-gray-600 font-bold">+</button>
                        </div>
                        <span class="text-xs font-bold text-gray-900 w-12 text-right">₹${itemTotal}</span>
                    </div>
                `;
                cartItemsContainer.appendChild(cartRow);
            });

            cartBadge.innerText = totalQty;
            cartBadge.style.transform = totalQty > 0 ? "scale(1)" : "scale(0)";
            cartSubtotal.innerText = `₹${total}`;
            cartTotal.innerText = `₹${total}`;
            cartTotalHeader.innerText = `₹${total}`;
            cartItemCount.innerText = totalQty;

            document.getElementById("checkout-btn").disabled = cart.length === 0;
        }

        function changeQty(index, delta) {
            cart[index].qty += delta;
            if (cart[index].qty <= 0) {
                cart.splice(index, 1);
            }
            updateCartUI();
        }

        function toggleCartDrawer(open) {
            const backdrop = document.getElementById("cart-backdrop");
            const panel = document.getElementById("cart-panel");
            if (open) {
                backdrop.classList.remove("hidden");
                setTimeout(() => backdrop.classList.remove("opacity-0"), 10);
                panel.classList.remove("translate-x-full");
            } else {
                backdrop.classList.add("opacity-0");
                panel.classList.add("translate-x-full");
                setTimeout(() => backdrop.classList.add("hidden"), 300);
            }
        }

        function openModal(id) { document.getElementById(id).classList.remove("hidden"); }
        function closeModal(id) { document.getElementById(id).classList.add("hidden"); }

        function openCheckoutModal() {
            toggleCartDrawer(false);
            openModal("checkout-modal");
        }

        function processOrder(e) {
            e.preventDefault();
            const name = document.getElementById("cust-name").value;
            const phone = document.getElementById("cust-phone").value;
            const address = document.getElementById("cust-address").value;

            let message = `*New Order from MeatFresh Website*\n\n`;
            message += `*Customer Details:*\nName: ${name}\nPhone: ${phone}\nAddress: ${address}\n\n`;
            message += `*Items Ordered:*\n`;

            let total = 0;
            cart.forEach((item, i) => {
                const sub = item.price * item.qty;
                total += sub;
                message += `${i + 1}. ${item.name} (${item.weight >= 1 ? item.weight + 'kg' : (item.weight * 1000) + 'g'}) x ${item.qty} = ₹${sub}\n`;
            });

            message += `\n*Total Payable:* ₹${total}\n`;
            message += `*Delivery Mode:* Express 30-Min Delivery`;

            const encoded = encodeURIComponent(message);
            window.open(`https://wa.me/${WHATSAPP_NUMBER}?text=${encoded}`, '_blank');
            closeModal("checkout-modal");
            cart = [];
            updateCartUI();
        }
